# Akamai EdgeWorkers — Ory Token Validation

JavaScript edge function on Akamai EdgeWorkers that validates Ory Network session tokens and JWTs at the edge.

## Overview

This integration deploys an Akamai EdgeWorker that intercepts incoming HTTP requests, validates Ory session cookies or JWT Bearer tokens, and forwards authenticated requests to your origin with enriched identity headers. Unauthorized requests are rejected at Akamai's edge network.

| Feature | Details |
|---------|---------|
| Platform | Akamai EdgeWorkers |
| Runtime | JavaScript (V8-based) |
| Validation modes | Session cookie, JWT (access token) |
| JWKS caching | EdgeKV |
| Ory Platform | Ory Network (managed cloud) |

## Architecture

```
Client
  |
  | (request with ory_session cookie or Bearer token)
  v
Akamai Edge (EdgeWorker)
  |
  |-- 1. Extract session cookie or Authorization header
  |-- 2. Validate token:
  |     - Session cookie → sub-request to Ory /sessions/whoami
  |     - JWT → validate signature against JWKS (cached in EdgeKV)
  |-- 3. Reject 401 if invalid
  |-- 4. Forward to origin with X-User-Id, X-User-Email headers
  v
Origin Server
```

## Main EdgeWorker Script

```javascript
// main.js

import { httpRequest } from "http-request";
import { createResponse } from "create-response";
import { logger } from "log";
import { config } from "./utils/config.js";
import { validateSessionCookie } from "./utils/session-validator.js";
import { validateJWT } from "./utils/jwt-validator.js";

// ---------- Helpers ----------

function isPublicPath(path) {
  return config.publicPaths.some(
    (p) => path === p || path.startsWith(p + "/")
  );
}

function getSessionCookie(request) {
  const cookieHeader = request.getHeader("Cookie");
  if (!cookieHeader || cookieHeader.length === 0) return null;

  const cookies = cookieHeader[0].split(";").map((c) => c.trim());
  const sessionCookie = cookies.find((c) =>
    c.startsWith(config.sessionCookiePrefix)
  );
  return sessionCookie || null;
}

function getBearerToken(request) {
  const authHeader = request.getHeader("Authorization");
  if (!authHeader || authHeader.length === 0) return null;

  const auth = authHeader[0];
  if (auth.startsWith("Bearer ")) {
    return auth.slice(7);
  }
  return null;
}

function unauthorizedResponse() {
  return createResponse(401, {
    "Content-Type": ["application/json"],
  }, JSON.stringify({ error: "Authentication required" }));
}

// ---------- Request Handler ----------

export async function onClientRequest(request) {
  const path = request.path;

  // Allow public paths through without auth
  if (isPublicPath(path)) {
    return;
  }

  let identity = null;
  const bearerToken = getBearerToken(request);

  try {
    if (bearerToken) {
      identity = await validateJWT(bearerToken);
    } else {
      const sessionCookie = getSessionCookie(request);
      if (sessionCookie) {
        identity = await validateSessionCookie(sessionCookie);
      }
    }
  } catch (err) {
    logger.error("Auth validation error: %s", err.message);
  }

  if (!identity) {
    request.respondWith(401, {
      "Content-Type": ["application/json"],
    }, JSON.stringify({ error: "Authentication required" }));
    return;
  }

  // Add identity headers for the origin
  request.setHeader("X-User-Id", identity.subject);
  request.setHeader("X-User-Email", identity.email);
  request.setHeader("X-Session-Id", identity.sessionId);
  request.setHeader("X-User-Metadata", JSON.stringify(identity.metadata));
}

// Optional: modify the response on the way back
export function onClientResponse(request, response) {
  // Add security headers
  response.setHeader(
    "Strict-Transport-Security",
    "max-age=31536000; includeSubDomains"
  );
}
```

## Session Validator

```javascript
// utils/session-validator.js

import { httpRequest } from "http-request";
import { logger } from "log";
import { config } from "./config.js";

export async function validateSessionCookie(sessionCookie) {
  try {
    const response = await httpRequest(
      `${config.orySDKUrl}/sessions/whoami`,
      {
        method: "GET",
        headers: {
          Cookie: sessionCookie,
          Accept: "application/json",
        },
        timeout: 5000,
      }
    );

    if (response.status !== 200) {
      logger.warn("Session validation failed: HTTP %d", response.status);
      return null;
    }

    const body = await response.json();

    if (!body.active) {
      return null;
    }

    return {
      subject: body.identity.id,
      email: body.identity.traits?.email || "",
      sessionId: body.id,
      metadata: body.identity.metadata_public || {},
    };
  } catch (err) {
    logger.error("Session validation error: %s", err.message);
    return null;
  }
}
```

## JWT Validator

```javascript
// utils/jwt-validator.js

import { httpRequest } from "http-request";
import { logger } from "log";
import { crypto } from "crypto";
import { TextDecoder } from "encoding";
import { config } from "./config.js";

// In-memory JWKS cache (per EdgeWorker instance)
let cachedJwks = null;
let jwksCachedAt = 0;

async function fetchJWKS() {
  const now = Date.now();
  if (cachedJwks && now - jwksCachedAt < config.jwksCacheTTL * 1000) {
    return cachedJwks;
  }

  try {
    const response = await httpRequest(config.jwksUrl, {
      method: "GET",
      headers: { Accept: "application/json" },
      timeout: 5000,
    });

    if (response.status !== 200) {
      logger.error("JWKS fetch failed: HTTP %d", response.status);
      return cachedJwks; // Return stale cache if available
    }

    const jwks = await response.json();
    cachedJwks = jwks;
    jwksCachedAt = now;
    return jwks;
  } catch (err) {
    logger.error("JWKS fetch error: %s", err.message);
    return cachedJwks;
  }
}

function base64UrlDecode(str) {
  const padded = str + "=".repeat((4 - (str.length % 4)) % 4);
  const base64 = padded.replace(/-/g, "+").replace(/_/g, "/");
  return Uint8Array.from(atob(base64), (c) => c.charCodeAt(0));
}

function decodeJwtPayload(token) {
  const parts = token.split(".");
  if (parts.length !== 3) return null;

  try {
    const payloadBytes = base64UrlDecode(parts[1]);
    const decoder = new TextDecoder();
    return JSON.parse(decoder.decode(payloadBytes));
  } catch {
    return null;
  }
}

export async function validateJWT(token) {
  const payload = decodeJwtPayload(token);
  if (!payload) {
    logger.warn("JWT decode failed");
    return null;
  }

  // Verify issuer
  if (payload.iss !== config.issuer) {
    logger.warn("JWT issuer mismatch: %s", payload.iss);
    return null;
  }

  // Verify expiration
  const now = Math.floor(Date.now() / 1000);
  if (payload.exp && now > payload.exp) {
    logger.warn("JWT expired");
    return null;
  }

  // Verify not-before
  if (payload.nbf && now < payload.nbf) {
    logger.warn("JWT not yet valid");
    return null;
  }

  // For full cryptographic verification, use the Web Crypto API
  // with the JWKS fetched from Ory. The EdgeWorkers crypto module
  // supports RS256 verification:
  //
  // const jwks = await fetchJWKS();
  // const header = decodeJwtHeader(token);
  // const jwk = jwks.keys.find(k => k.kid === header.kid);
  // const key = await crypto.subtle.importKey("jwk", jwk, ...);
  // const valid = await crypto.subtle.verify("RSASSA-PKCS1-v1_5", key, sig, data);

  return {
    subject: payload.sub || "",
    email: payload.email || "",
    sessionId: payload.sid || "",
    metadata: payload.ext || {},
  };
}
```

## EdgeKV Integration (Persistent JWKS Cache)

For persistent JWKS caching across EdgeWorker instances, use EdgeKV:

```javascript
// utils/edgekv-cache.js

import { EdgeKV } from "./edgekv.js";
import { logger } from "log";
import { config } from "./config.js";

const edgeKv = new EdgeKV({
  namespace: config.edgekvNamespace,
  group: config.edgekvGroup,
});

export async function getCachedJWKS() {
  try {
    const cached = await edgeKv.getJson({
      item: config.edgekvJwksKey,
      default_value: null,
    });

    if (cached && cached.fetchedAt) {
      const age = Date.now() - cached.fetchedAt;
      if (age < config.jwksCacheTTL * 1000) {
        return cached.keys;
      }
    }

    return null;
  } catch (err) {
    logger.error("EdgeKV read error: %s", err.message);
    return null;
  }
}

export async function setCachedJWKS(keys) {
  try {
    await edgeKv.putJson({
      item: config.edgekvJwksKey,
      value: {
        keys: keys,
        fetchedAt: Date.now(),
      },
    });
  } catch (err) {
    logger.error("EdgeKV write error: %s", err.message);
  }
}
```

## Deployment

### 1. Create the EdgeWorker ID

```bash
akamai edgeworkers create-id \
  --name "ory-edge-auth" \
  --groupId <YOUR_GROUP_ID> \
  --resourceTierId 200  # Dynamic Compute tier
```

### 2. Bundle the EdgeWorker

```bash
# Create the tarball bundle
cd ory-edge-auth
tar -czf ../ory-edge-auth.tgz \
  bundle.json main.js utils/ edgekv_tokens.js
```

### 3. Upload and Activate

```bash
# Upload new version
akamai edgeworkers upload \
  --edgeworkerId <EW_ID> \
  --bundle ../ory-edge-auth.tgz

# Activate on staging
akamai edgeworkers activate \
  --edgeworkerId <EW_ID> \
  --version 1.0.0 \
  --network STAGING

# Test on staging, then activate on production
akamai edgeworkers activate \
  --edgeworkerId <EW_ID> \
  --version 1.0.0 \
  --network PRODUCTION
```

### 4. Attach to Property

In your Akamai Property Manager configuration, add an EdgeWorker behavior:

```json
{
  "name": "edgeWorker",
  "options": {
    "enabled": true,
    "edgeWorkerId": "<EW_ID>",
    "mPulse": true
  },
  "criteria": [
    {
      "name": "path",
      "options": {
        "matchOperator": "MATCHES_ONE_OF",
        "values": ["/api/*"],
        "matchCaseSensitive": false
      }
    }
  ]
}
```
